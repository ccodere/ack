cmake_minimum_required (VERSION 3.0)
project(ego)

set(SRC_C
	 debug.c
	 global.c
	 files.c
     go.c
	 map.c
	 utils.c
	 get.c
	 put.c
     alloc.c
	 lset.c
	 cset.c
	 parser.c
	 stack_chg.c
     locals.c
     init_glob.c
)

set(GEN_SRC 
 ${CMAKE_CURRENT_BINARY_DIR}/classdefs.h
 ${CMAKE_CURRENT_BINARY_DIR}/pop_push.h
)

set(INCLUDE_DIRS
 .
 ${CMAKE_CURRENT_BINARY_DIR}/../../../h
 ${CMAKE_CURRENT_BINARY_DIR}
)


add_library(${PROJECT_NAME} ${SRC_C} ${GEN_SRC})
target_include_directories(${PROJECT_NAME} PUBLIC ${INCLUDE_DIRS})
target_compile_definitions(${PROJECT_NAME} PUBLIC NORCSID=1 DEBUG VERBOSE)
target_link_libraries(${PROJECT_NAME} emheaders em_data)


find_program (AWK NAMES awk gawk)

if(NOT AWK)
message(SEND_ERROR "awk tool not found to build \"${PROJECT_NAME}\"")
endif(NOT AWK)

add_custom_command(
  OUTPUT pop_push.h
  COMMAND ${CMAKE_COMMAND} -E copy
                ${CMAKE_CURRENT_SOURCE_DIR}/pop_push.awk
                ${CMAKE_CURRENT_BINARY_DIR}/pop_push.awk
  COMMAND ${AWK} -f ${CMAKE_CURRENT_BINARY_DIR}/pop_push.awk ${CMAKE_CURRENT_SOURCE_DIR}/../../../h/em_table>pop_push.h
  COMMENT "Generating pop_push.h files"
)



install(TARGETS ${PROJECT_NAME} 
        RUNTIME DESTINATION bin
        LIBRARY DESTINATION lib
        ARCHIVE DESTINATION lib
        PUBLIC_HEADER DESTINATION include
)
